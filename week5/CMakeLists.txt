cmake_minimum_required(VERSION 3.16)
project(external_storage_week5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(run_storage src/buffer.cpp src/run_file.cpp src/sort.cpp src/advanced.cpp src/replacement.cpp src/pipeline.cpp)
target_link_libraries(run_storage PUBLIC Threads::Threads)
target_include_directories(run_storage PUBLIC include)
if(MSVC)
  target_compile_options(run_storage PUBLIC /W4 /utf-8)
else()
  target_compile_options(run_storage PUBLIC -Wall -Wextra -Wpedantic)
endif()
add_executable(week3_demo src/main.cpp)
target_link_libraries(week3_demo PRIVATE run_storage)
enable_testing()
add_executable(week3_tests tests/week3_tests.cpp)
target_link_libraries(week3_tests PRIVATE run_storage)
add_test(NAME week3_storage COMMAND week3_tests)
add_executable(week4_demo src/week4_main.cpp)
target_link_libraries(week4_demo PRIVATE run_storage)
add_executable(week4_tests tests/week4_tests.cpp)
target_link_libraries(week4_tests PRIVATE run_storage)
add_test(NAME week4_sort COMMAND week4_tests)
add_executable(week5_serial_tests tests/week5_serial_tests.cpp)
target_link_libraries(week5_serial_tests PRIVATE run_storage)
add_test(NAME week5_serial COMMAND week5_serial_tests)
add_executable(week5_demo src/week5_main.cpp)
target_link_libraries(week5_demo PRIVATE run_storage)
add_executable(week5_pipeline_tests tests/week5_pipeline_tests.cpp)
target_link_libraries(week5_pipeline_tests PRIVATE run_storage)
add_test(NAME week5_pipeline COMMAND week5_pipeline_tests)
set_tests_properties(week5_pipeline PROPERTIES TIMEOUT 120)
